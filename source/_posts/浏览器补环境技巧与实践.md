## 适用场景

扣出来的JS代码在Node.js里运行报错，提示`window is not defined`、`document is not defined`。需要补全浏览器环境，让JS代码以为自己在浏览器里运行。

## 实现原理

前端JS代码依赖浏览器提供的全局对象（window、document、navigator、location等），Node.js里没有这些对象。补环境就是在Node.js里模拟这些浏览器对象，让JS代码能正常运行。常用的补环境工具有jsdom、puppeteer等。

## 具体步骤

### 第一步：识别需要补的环境

运行扣出来的JS代码，看报错信息：

```
ReferenceError: window is not defined
ReferenceError: document is not defined
ReferenceError: navigator is not defined
ReferenceError: location is not defined
ReferenceError: localStorage is not defined
```

根据报错逐个补全。

### 第二步：基础补环境

```javascript
// 最简单的补环境
global.window = global;
global.document = {
    createElement: () => ({}),
    getElementById: () => null,
    getElementsByTagName: () => [],
    querySelector: () => null,
    querySelectorAll: () => [],
    addEventListener: () => {},
    cookie: "",
    referrer: "",
    title: ""
};
global.navigator = {
    userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    platform: "Win32",
    language: "zh-CN",
    appName: "Netscape",
    appVersion: "5.0"
};
global.location = {
    href: "https://example.com",
    host: "example.com",
    hostname: "example.com",
    protocol: "https:",
    pathname: "/",
    search: ""
};
```

### 第三步：用jsdom补环境（推荐）

jsdom是最常用的浏览器环境模拟库：

```bash
npm install jsdom
```

```javascript
const { JSDOM } = require("jsdom");

// 创建虚拟DOM
const dom = new JSDOM(`<!DOCTYPE html><html><head></head><body></body></html>`, {
    url: "https://example.com",
    referrer: "https://example.com",
    contentType: "text/html",
    userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    pretendToBeVisual: true
});

// 把jsdom的对象挂到全局
global.window = dom.window;
global.document = dom.window.document;
global.navigator = dom.window.navigator;
global.location = dom.window.location;
global.HTMLElement = dom.window.HTMLElement;
global.Node = dom.window.Node;

// 运行JS代码
const fs = require("fs");
const code = fs.readFileSync("spider.js", "utf-8");
eval(code);
```

### 第四步：补全常用对象

```javascript
// localStorage
global.localStorage = {
    _data: {},
    getItem(key) { return this._data[key] || null; },
    setItem(key, value) { this._data[key] = value; },
    removeItem(key) { delete this._data[key]; },
    clear() { this._data = {}; }
};

// sessionStorage
global.sessionStorage = { ...global.localStorage };

// console（Node.js已有，但有些代码需要特定格式）
global.console = console;

// XMLHttpRequest
global.XMLHttpRequest = require("xmlhttprequest").XMLHttpRequest;

// fetch
global.fetch = require("node-fetch");

// Canvas（有些代码会用canvas做指纹）
const { createCanvas } = require("canvas");
global.HTMLCanvasElement = class {
    getContext() {
        return createCanvas(200, 200).getContext("2d");
    }
};

// setTimeout/setInterval（Node.js已有）
// global.setTimeout = setTimeout;
// global.setInterval = setInterval;
```

### 第五步：补全canvas指纹

很多网站用canvas做浏览器指纹，需要补全：

```bash
npm install canvas
```

```javascript
const { createCanvas, Image } = require("canvas");

// 补全canvas
global.HTMLCanvasElement = class HTMLCanvasElement {
    constructor() {
        this._canvas = createCanvas(300, 150);
        this.width = 300;
        this.height = 150;
    }
    getContext(type) {
        return this._canvas.getContext(type);
    }
    toDataURL() {
        return this._canvas.toDataURL();
    }
};

global.Image = Image;
```

### 第六步：检测环境差异

有些代码会检测是不是真实浏览器，需要绕过：

```javascript
// 1. 检测window对象
// 真实浏览器中 window === globalThis
global.window = global;

// 2. 检测document.referrer
Object.defineProperty(global.document, "referrer", {
    value: "https://example.com",
    writable: false
});

// 3. 检测navigator.webdriver（自动化检测）
Object.defineProperty(global.navigator, "webdriver", {
    value: undefined,
    writable: false
});

// 4. 检测chrome对象
global.chrome = {
    runtime: {}
};

// 5. 检测permissions
global.navigator.permissions = {
    query: () => Promise.resolve({ state: "prompt" })
};
```

### 第七步：用puppeteer运行（终极方案）

如果补环境太麻烦，直接用puppeteer在真实浏览器里运行：

```bash
npm install puppeteer
```

```javascript
const puppeteer = require("puppeteer");

async function runInBrowser(jsCode) {
    const browser = await puppeteer.launch({ headless: true });
    const page = await browser.newPage();
    
    // 先访问目标网站，获取真实环境
    await page.goto("https://example.com");
    
    // 在页面上下文运行JS代码
    const result = await page.evaluate((code) => {
        return eval(code);
    }, jsCode);
    
    await browser.close();
    return result;
}
```

## 完整代码

```javascript
// env.js - 完整的补环境脚本
const { JSDOM } = require("jsdom");
const fs = require("fs");

function setupBrowserEnv(url = "https://example.com") {
    // 1. 创建jsdom环境
    const dom = new JSDOM(`<!DOCTYPE html><html><head></head><body></body></html>`, {
        url: url,
        referrer: url,
        userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        pretendToBeVisual: true
    });
    
    // 2. 挂载全局对象
    global.window = dom.window;
    global.document = dom.window.document;
    global.navigator = dom.window.navigator;
    global.location = dom.window.location;
    global.history = dom.window.history;
    global.HTMLElement = dom.window.HTMLElement;
    global.Node = dom.window.Node;
    global.Event = dom.window.Event;
    global.MouseEvent = dom.window.MouseEvent;
    global.KeyboardEvent = dom.window.KeyboardEvent;
    
    // 3. 补全localStorage
    global.localStorage = dom.window.localStorage;
    global.sessionStorage = dom.window.sessionStorage;
    
    // 4. 补全navigator属性
    Object.defineProperty(global.navigator, "webdriver", {
        value: undefined,
        writable: false
    });
    Object.defineProperty(global.navigator, "languages", {
        value: ["zh-CN", "zh", "en"],
        writable: false
    });
    Object.defineProperty(global.navigator, "plugins", {
        value: [1, 2, 3, 4, 5],
        writable: false
    });
    
    // 5. 补全chrome对象
    global.chrome = { runtime: {} };
    
    // 6. 补全XMLHttpRequest
    global.XMLHttpRequest = dom.window.XMLHttpRequest;
    
    // 7. 补全fetch
    global.fetch = dom.window.fetch;
    
    // 8. 补全canvas（可选）
    try {
        const { createCanvas, Image } = require("canvas");
        global.HTMLCanvasElement = class {
            constructor() {
                this._canvas = createCanvas(300, 150);
                this.width = 300;
                this.height = 150;
            }
            getContext(type) { return this._canvas.getContext(type); }
            toDataURL() { return this._canvas.toDataURL(); }
        };
        global.Image = Image;
    } catch (e) {
        console.log("canvas模块未安装，跳过canvas补全");
    }
    
    console.log("浏览器环境补全完成");
    return dom;
}

// 使用
setupBrowserEnv("https://example.com");

// 运行扣出来的JS代码
const code = fs.readFileSync("spider.js", "utf-8");
eval(code);
```

## 注意事项

1. **逐步补全** — 不要一次性补所有，运行报错了再补对应的对象
2. **jsdom有限制** — jsdom不是完整浏览器，canvas、WebGL等可能不支持
3. **环境检测** — 有些代码会检测环境差异，需要针对性绕过
4. **puppeteer最稳** — 补环境太麻烦时，直接用puppeteer在真实浏览器运行
5. **版本兼容** — 注意jsdom和Node.js版本的兼容性
