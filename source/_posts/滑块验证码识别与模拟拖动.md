## 适用场景

登录或请求时出现滑块验证码，需要拖动滑块到缺口位置。包括简单的滑块拼图和复杂的极验、腾讯防水墙等行为验证。

## 实现原理

滑块验证码通过两个维度验证：1）缺口位置是否正确（图像识别）；2）拖动轨迹是否像人（行为检测）。爬虫需要用图像识别找到缺口位置，然后模拟人类的拖动轨迹（先快后慢、有抖动），而不是匀速直线移动。

## 具体步骤

### 第一步：滑块验证码类型

1. **简单滑块** — 只有一个滑块和缺口，识别缺口位置即可
2. **极验滑块** — 有背景图和滑块图，需要对比找缺口，还有行为检测
3. **腾讯防水墙** — 类似极验，有旋转、点选等多种类型
4. **网易易盾** — 滑块+点选组合

### 第三步：识别缺口位置

**方法1：像素对比（简单滑块）**
```python
from PIL import Image

def find_gap(bg_path, slider_path):
    """对比背景图和滑块图，找缺口位置"""
    bg = Image.open(bg_path).convert("RGB")
    slider = Image.open(slider_path).convert("RGB")
    
    bg_pixels = bg.load()
    slider_pixels = slider.load()
    
    width, height = bg.size
    
    # 遍历像素，找差异最大的位置
    for x in range(width):
        for y in range(height):
            bg_r, bg_g, bg_b = bg_pixels[x, y]
            s_r, s_g, s_b = slider_pixels[x, y] if x < slider.size[0] else (0, 0, 0)
            
            # 像素差异大的地方就是缺口
            if abs(bg_r - s_r) > 50 and abs(bg_g - s_g) > 50 and abs(bg_b - s_b) > 50:
                return x  # 返回缺口的x坐标
    
    return 0
```

**方法2：OpenCV边缘检测（推荐）**
```python
import cv2
import numpy as np

def find_gap_opencv(bg_path, slider_path):
    """用OpenCV识别缺口位置"""
    bg = cv2.imread(bg_path, 0)  # 灰度图
    slider = cv2.imread(slider_path, 0)
    
    # 边缘检测
    bg_edge = cv2.Canny(bg, 100, 200)
    slider_edge = cv2.Canny(slider, 100, 200)
    
    # 模板匹配
    result = cv2.matchTemplate(bg_edge, slider_edge, cv2.TM_CCOEFF_NORMED)
    min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)
    
    # max_loc[0]就是缺口的x坐标
    return max_loc[0]
```

### 第四步：生成人类拖动轨迹

```python
import random
import numpy as np

def generate_track(distance):
    """生成人类的拖动轨迹"""
    tracks = []
    current = 0
    mid = distance * 4 / 5  # 前4/5加速，后1/5减速
    t = 0.2  # 时间间隔
    v = 0    # 初速度
    
    while current < distance:
        if current < mid:
            a = random.randint(2, 4)  # 加速
        else:
            a = -random.randint(2, 4)  # 减速
        
        v0 = v
        v = v0 + a * t
        move = v0 * t + 0.5 * a * t * t
        current += move
        tracks.append(round(move))
    
    # 最后加一些小抖动
    for _ in range(random.randint(3, 5)):
        tracks.append(random.randint(-2, 2))
    
    return tracks
```

### 第五步：Selenium模拟拖动

```python
from selenium import webdriver
from selenium.webdriver.common.action_chains import ActionChains
import time

def slide_captcha(driver, slider_element, distance):
    """用Selenium拖动滑块"""
    # 生成轨迹
    tracks = generate_track(distance)
    
    # 按下滑块
    ActionChains(driver).click_and_hold(slider_element).perform()
    time.sleep(0.2)
    
    # 按轨迹拖动
    for track in tracks:
        ActionChains(driver).move_by_offset(xoffset=track, yoffset=0).perform()
        time.sleep(random.uniform(0.01, 0.03))
    
    # 小范围抖动
    time.sleep(0.3)
    ActionChains(driver).move_by_offset(xoffset=-3, yoffset=0).perform()
    time.sleep(0.1)
    ActionChains(driver).move_by_offset(xoffset=3, yoffset=0).perform()
    
    # 释放滑块
    time.sleep(0.2)
    ActionChains(driver).release().perform()
```

```python
def slide_captcha_playwright(page, slider_selector, distance):
    """用Playwright拖动滑块"""
    slider = page.query_selector(slider_selector)
    box = slider.bounding_box()
    
    # 鼠标移动到滑块中心

    page.mouse.down()
    
    # 生成轨迹并拖动
    tracks = generate_track(distance)
    current_x = box["x"] + box["width"] / 2
    
    for track in tracks:
        current_x += track
        page.mouse.move(current_x, box["y"] + box["height"] / 2)
        page.wait_for_timeout(random.randint(10, 30))
    
    # 释放
    page.mouse.up()
```

## 完整代码

```python
import cv2
import numpy as np
import random
import time
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.action_chains import ActionChains

class SliderCaptcha:
    def __init__(self, driver):
        self.driver = driver
    
    def find_gap(self, bg_path, slider_path):
        """识别缺口位置"""
        bg = cv2.imread(bg_path, 0)
        slider = cv2.imread(slider_path, 0)
        
        bg_edge = cv2.Canny(bg, 100, 200)
        slider_edge = cv2.Canny(slider, 100, 200)
        
        result = cv2.matchTemplate(bg_edge, slider_edge, cv2.TM_CCOEFF_NORMED)
        _, _, _, max_loc = cv2.minMaxLoc(result)
        return max_loc[0]
    
    def generate_track(self, distance):
        """生成拖动轨迹"""
        tracks = []
        current = 0
        mid = distance * 4 / 5
        t = 0.2
        v = 0
        
        while current < distance:
            a = random.randint(2, 4) if current < mid else -random.randint(2, 4)
            v0 = v
            v = v0 + a * t
            move = v0 * t + 0.5 * a * t * t
            current += move
            tracks.append(round(move))
        
        # 回退抖动
        tracks.extend([-1, -1, -2, 1, 2, 1])
        return tracks
    
    def slide(self, slider_selector, bg_url, slider_url):
        """完整的滑块验证流程"""
        # 1. 下载图片
        bg_img = self.driver.find_element(By.CSS_SELECTOR, ".captcha-bg").get_attribute("src")
        slider_img = self.driver.find_element(By.CSS_SELECTOR, ".captcha-slider").get_attribute("src")
        
        # 2. 识别缺口
        distance = self.find_gap("bg.png", "slider.png")
        print(f"缺口位置: {distance}")
        
        # 3. 生成轨迹
        tracks = self.generate_track(distance)
        
        # 4. 拖动
        slider = self.driver.find_element(By.CSS_SELECTOR, slider_selector)
        ActionChains(self.driver).click_and_hold(slider).perform()
        time.sleep(0.2)
        
        for track in tracks:
            ActionChains(self.driver).move_by_offset(xoffset=track, yoffset=0).perform()
            time.sleep(random.uniform(0.01, 0.03))
        
        time.sleep(0.3)
        ActionChains(self.driver).release().perform()
        time.sleep(1)
        
        # 5. 检查是否成功
        return "验证成功" in self.driver.page_source

if __name__ == "__main__":
    driver = webdriver.Chrome()
    driver.get("https://example.com/login")
    
    captcha = SliderCaptcha(driver)
    success = captcha.slide(".slider-btn", "bg.png", "slider.png")
    print(f"验证结果: {success}")
```

## 注意事项

1. **图片缩放** — 网页显示的图片和实际图片尺寸可能不一样，要计算缩放比例
2. **轨迹很重要** — 匀速移动一定会被检测，必须模拟人类的加速减速和抖动
3. **极验更复杂** — 极验3.0/4.0有更复杂的行为检测，简单轨迹模拟可能过不了
4. **打码平台** — 复杂的滑块验证码可以用打码平台，返回缺口坐标
5. **失败重试** — 滑块验证可能失败，要做好重试机制
