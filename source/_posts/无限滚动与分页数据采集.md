## 适用场景

页面没有下一页按钮，滚动到底部自动加载更多内容，比如微博、知乎、小红书、抖音网页版。需要采集大量列表数据。

## 实现原理

无限滚动的本质还是分页，只是用JS监听滚动事件，滚动到底部时自动请求下一页数据，追加到页面上。爬虫可以直接分析分页API，或者用浏览器模拟滚动。

## 具体步骤

### 第一步：分析分页API（推荐）

无限滚动一般都有分页接口，直接调用API最快：

1. F12 → Network → Fetch/XHR
2. 滚动页面，观察新出现的请求
3. 对比多次请求的URL和参数
4. 找出分页参数的规律

常见分页参数：
- `page=1&page_size=20`
- `offset=0&limit=20`
- `cursor=xxx&count=20`
- `max_id=xxx&since_id=xxx`
- `page_token=xxx`

### 第二步：普通分页采集

```python
import requests

def fetch_by_page():
    """普通分页采集"""
    base_url = "https://api.example.com/feed"
    headers = {"User-Agent": "Mozilla/5.0"}
    
    all_data = []
    page = 1
    
    while True:
        params = {"page": page, "page_size": 20}
        response = requests.get(base_url, headers=headers, params=params)
        data = response.json()
        
        items = data.get("data", {}).get("list", [])
        if not items:
            break  # 没有数据了，结束
        
        all_data.extend(items)
        print(f"第{page}页，获取{len(items)}条，累计{len(all_data)}条")
        
        # 判断是否还有下一页
        if not data.get("data", {}).get("has_more", False):
            break
        
        page += 1
    
    return all_data
```

### 第三步：cursor/max_id分页

有些网站用cursor或max_id分页，下一页的参数来自上一页的响应：

```python
def fetch_by_cursor():
    """cursor分页采集"""
    base_url = "https://api.example.com/feed"
    headers = {"User-Agent": "Mozilla/5.0"}
    
    all_data = []
    cursor = None  # 第一页没有cursor
    
    while True:
        params = {"count": 20}
        if cursor:
            params["cursor"] = cursor
        
        response = requests.get(base_url, headers=headers, params=params)
        data = response.json()
        
        items = data.get("items", [])
        if not items:
            break
        
        all_data.extend(items)
        print(f"获取{len(items)}条，累计{len(all_data)}条")
        
        # 从响应中获取下一页的cursor
        cursor = data.get("next_cursor")
        if not cursor or not data.get("has_more", False):
            break
    
    return all_data
```

### 第四步：用浏览器模拟滚动

找不到API或者API有签名加密时，用浏览器模拟滚动：

```python
from playwright.sync_api import sync_playwright

def fetch_by_scroll(url, max_scrolls=50):
    """用浏览器模拟滚动采集"""
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        page.goto(url, wait_until="networkidle")
        
        all_items = []
        last_count = 0
        
        for i in range(max_scrolls):
            # 滚动到底部
            page.evaluate("window.scrollTo(0, document.body.scrollHeight)")
            page.wait_for_timeout(2000)  # 等待加载
            
            # 提取当前页面所有数据
            items = page.query_selector_all(".item")
            current_count = len(items)
            
            if current_count == last_count:
                print("没有新数据，加载完成")
                break
            
            print(f"第{i+1}次滚动，当前{current_count}条")
            last_count = current_count
        
        # 提取所有数据
        for item in page.query_selector_all(".item"):
            title = item.query_selector(".title").text_content()
            all_items.append(title)
        
        browser.close()
        return all_items
```

### 第五步：监听网络请求（推荐）

用Playwright监听网络请求，直接捕获API响应，比解析HTML快：

```python
def fetch_by_network_listen(url):
    """监听网络请求，直接捕获API数据"""
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        
        api_data = []
        
        def handle_response(response):
            if "/api/feed" in response.url:
                try:
                    data = response.json()
                    api_data.extend(data.get("items", []))
                except:
                    pass
        
        page.on("response", handle_response)
        page.goto(url, wait_until="networkidle")
        
        # 滚动加载更多
        for _ in range(10):
            page.evaluate("window.scrollTo(0, document.body.scrollHeight)")
            page.wait_for_timeout(2000)
        
        browser.close()
        return api_data
```

## 完整代码

```python
import requests
import time
import random

class InfiniteScrollSpider:
    def __init__(self):
        self.headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
            "Accept": "application/json"
        }
        self.session = requests.Session()
    
    def fetch_api(self, url, params):
        """请求API，带重试"""
        for retry in range(3):
            try:
                response = self.session.get(
                    url, headers=self.headers, params=params, timeout=10
                )
                response.raise_for_status()
                return response.json()
            except Exception as e:
                print(f"请求失败，重试{retry+1}: {e}")
                time.sleep(random.uniform(1, 3))
        return None
    
    def crawl_by_page(self, base_url, max_pages=100):
        """普通分页采集"""
        all_data = []
        for page in range(1, max_pages + 1):
            data = self.fetch_api(base_url, {"page": page, "page_size": 20})
            if not data:
                break
            
            items = data.get("data", {}).get("list", [])
            if not items:
                break
            
            all_data.extend(items)
            print(f"第{page}页：{len(items)}条，累计{len(all_data)}条")
            
            if not data.get("data", {}).get("has_more"):
                break
            
            time.sleep(random.uniform(0.5, 2))
        
        return all_data
    
    def crawl_by_cursor(self, base_url, max_rounds=100):
        """cursor分页采集"""
        all_data = []
        cursor = None
        
        for _ in range(max_rounds):
            params = {"count": 20}
            if cursor:
                params["cursor"] = cursor
            
            data = self.fetch_api(base_url, params)
            if not data:
                break
            
            items = data.get("items", [])
            if not items:
                break
            
            all_data.extend(items)
            print(f"获取{len(items)}条，累计{len(all_data)}条")
            
            cursor = data.get("next_cursor")
            if not cursor or not data.get("has_more"):
                break
            
            time.sleep(random.uniform(0.5, 2))
        
        return all_data

if __name__ == "__main__":
    spider = InfiniteScrollSpider()
    data = spider.crawl_by_page("https://api.example.com/feed")
    print(f"总共获取{len(data)}条数据")
```

## 注意事项

1. **优先找API** — 无限滚动一定有API，直接调用比滚动快得多
2. **分页参数** — 注意区分page/offset/cursor/max_id等不同分页方式
3. **去重** — 滚动加载可能有重复数据，要根据ID去重
4. **频率控制** — 不要请求太快，容易触发限流
5. **终止条件** — 一定要有终止条件，不然可能无限循环
